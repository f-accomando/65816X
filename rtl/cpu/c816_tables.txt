# c816_tables.txt - the only source of c816_decode and c816_ucode.
# Format and codes: docs/design/cpu-core.md, sections 4 and 5. After a change run: make tables

#   t bus asel cap pc  sop wsel last skip pcop hold
[seq IMP]                       # implied: 2 cycles
    1 IO  PC   -   -   -   -    ALW  -    -    -
[seq WDM]                       # WDM, ext_en = 0: 2 cycles, the second byte is skipped
    1 IO  PC   -   INC -   -    ALW  -    -    -
[seq WDMX]                      # WDM, ext_en = 1: the sub-opcode is a program byte
    1 PRG PC   DL  INC -   -    ALW  -    -    -
[seq IMM]                       # immediate: 2 (8-bit) or 3 (16-bit)
    1 PRG PC   DL  INC -   -    W8   -    -    -
    2 PRG PC   DH  INC -   -    ALW  -    -    -
[seq IMM8P]                     # REP, SEP: 3
    1 PRG PC   DL  INC -   -    -    -    -    -
    2 IO  PCD  -   -   -   -    ALW  -    -    -
[seq ABS]                       # 4 or 5
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 DAT EA   DL  -   -   RLO  W8   -    -    -
    4 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq ABSX]                      # load 4-6, store 5-6
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    XNC  -    -
    3 IO  DUMI -   -   -   -    -    -    -    -
    4 DAT EA   DL  -   -   RLO  W8   -    -    -
    5 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq LONG]                      # 5 or 6
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 PRG PC   BA  INC -   -    -    -    -    -
    4 DAT EA   DL  -   -   RLO  W8   -    -    -
    5 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq JMP]                       # 3
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    ALW  -    ABS  -
[seq JML]                       # 4
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 PRG PC   BA  INC -   -    ALW  -    LONG -
[seq BR]                        # 2 not taken, 3 taken, 4 taken with E = 1 and page cross
    1 PRG PC   AL  INC -   -    NTK  -    -    -
    2 IO  PCD  -   -   -   -    NEC  -    BRL  -
    3 IO  PCD  -   -   -   -    ALW  -    BR   -
[seq JSR]                       # 6
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DWR S    -   -   DEC PCDH -    -    -    -
    5 DWR S    -   -   DEC PCDL ALW  -    ABS  -
[seq RTS]                       # 6
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 DRD S1   AL  -   INC -    -    -    -    -
    4 DRD S1   AH  -   INC -    -    -    -    -
    5 IO  S    -   -   -   -    ALW  -    RTS  -
[seq PUSH]                      # PHA: 3 or 4, high byte first
    1 IO  PC   -   -   -   -    -    W8   -    -
    2 DWR S    -   -   DEC RHI  -    -    -    -
    3 DWR S    -   -   DEC RLO  ALW  -    -    -
[seq PULL]                      # PLA: 4 or 5, low byte first
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 DRD S1   DL  -   INC -    W8   -    -    -
    4 DRD S1   DH  -   INC -    ALW  -    -    -
[seq RST]                       # pseudo-op 100: reset, three stack reads without writes
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 DRD S    -   -   DEC -    -    -    -    -
    4 DRD S    -   -   DEC -    -    -    -    -
    5 DRD S    -   -   DEC -    -    -    -    -
    6 DRD VEC  AL  -   -   -    -    -    -    -
    7 DRD VEC1 AH  -   -   -    ALW  -    VECT -

[seq DP]                        # direct page: 3 to 5, one more cycle if D[7:0] != 0
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DAT EA   DL  -   -   RLO  W8   -    -    -
    4 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPX]                       # direct page,X and ,Y
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DAT EA   DL  -   -   RLO  W8   -    -    -
    5 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPIND]                     # (dp)
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD DP0  AL  -   -   -    -    -    -    -
    4 DRD DP1  AH  -   -   -    -    -    -    -
    5 DAT EA   DL  -   -   RLO  W8   -    -    -
    6 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPXIND]                    # (dp,X)
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DRD DX0  AL  -   -   -    -    -    -    -
    5 DRD DX1  AH  -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPINDY]                    # (dp),Y: the dummy cycle as in abs,X
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD DP0  AL  -   -   -    -    -    -    -
    4 DRD DP1  AH  -   -   -    -    XNC  -    -
    5 IO  DUMI -   -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPLONG]                    # [dp] and [dp],Y
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD LP0  AL  -   -   -    -    -    -    -
    4 DRD LP1  AH  -   -   -    -    -    -    -
    5 DRD LP2  BA  -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq SR]                        # stack relative
    1 PRG PC   OF  INC -   -    -    -    -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DAT EA   DL  -   -   RLO  W8   -    -    -
    4 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq SRINDY]                    # (sr,S),Y
    1 PRG PC   OF  INC -   -    -    -    -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD SR0  AL  -   -   -    -    -    -    -
    4 DRD SR1  AH  -   -   -    -    -    -    -
    5 IO  SR1  -   -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -

# read-modify-write: read low (and high), modify cycle (a write of the old byte in emulation
# mode), write high, write low
[seq RMWDP]
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD EA   DL  -   -   -    -    W8   -    -
    4 DRD EA1  DH  -   -   -    -    -    -    -
    5 MOD EAL  -   -   -   DL   -    W8   -    -
    6 DWR EA1  -   -   -   ALUH -    -    -    -
    7 DWR EA   -   -   -   ALUL ALW  -    -    -
[seq RMWDPX]
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DRD EA   DL  -   -   -    -    W8   -    -
    5 DRD EA1  DH  -   -   -    -    -    -    -
    6 MOD EAL  -   -   -   DL   -    W8   -    -
    7 DWR EA1  -   -   -   ALUH -    -    -    -
    8 DWR EA   -   -   -   ALUL ALW  -    -    -
[seq RMWABS]
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 DRD EA   DL  -   -   -    -    W8   -    -
    4 DRD EA1  DH  -   -   -    -    -    -    -
    5 MOD EAL  -   -   -   DL   -    W8   -    -
    6 DWR EA1  -   -   -   ALUH -    -    -    -
    7 DWR EA   -   -   -   ALUL ALW  -    -    -
[seq RMWABSX]
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 IO  DUMI -   -   -   -    -    -    -    -
    4 DRD EA   DL  -   -   -    -    W8   -    -
    5 DRD EA1  DH  -   -   -    -    -    -    -
    6 MOD EAL  -   -   -   DL   -    W8   -    -
    7 DWR EA1  -   -   -   ALUH -    -    -    -
    8 DWR EA   -   -   -   ALUL ALW  -    -    -
[seq XBA]
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    ALW  -    -    -
# instructions of the 65816 on the stack: S does not stay in page 1 until the last cycle
[seq PUSHW]                     # PHD
    1 IO  PC   -   -   -   -    -    -    -    WIDE
    2 DWR SW   -   -   DEC RHI  -    -    -    WIDE
    3 DWR SW   -   -   DEC RLO  ALW  -    -    WIDE
[seq PULLW]                     # PLD, PLB
    1 IO  PC   -   -   -   -    -    -    -    WIDE
    2 IO  PC   -   -   -   -    -    -    -    WIDE
    3 DRD S1W  DL  -   INC -    W8   -    -    WIDE
    4 DRD S1W  DH  -   INC -    ALW  -    -    WIDE
[seq PEA]
    1 PRG PC   AL  INC -   -    -    -    -    WIDE
    2 PRG PC   AH  INC -   -    -    -    -    WIDE
    3 DWR SW   -   -   DEC AH   -    -    -    WIDE
    4 DWR SW   -   -   DEC AL   ALW  -    -    WIDE
[seq PEI]
    1 PRG PC   OF  INC -   -    -    DL0  -    WIDE
    2 IO  PCD  -   -   -   -    -    -    -    WIDE
    3 DRD LP0  AL  -   -   -    -    -    -    WIDE
    4 DRD LP1  AH  -   -   -    -    -    -    WIDE
    5 DWR SW   -   -   DEC AH   -    -    -    WIDE
    6 DWR SW   -   -   DEC AL   ALW  -    -    WIDE
[seq PER]
    1 PRG PC   AL  INC -   -    -    -    -    WIDE
    2 PRG PC   AH  INC -   -    -    -    -    WIDE
    3 IO  PCD  -   -   -   -    -    -    -    WIDE
    4 DWR SW   -   -   DEC PERH -    -    -    WIDE
    5 DWR SW   -   -   DEC PERL ALW  -    -    WIDE
[seq BRL]
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    ALW  -    REL16 -
[seq JMPI]                      # JMP (abs)
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 DRD AB0  DL  -   -   -    -    -    -    -
    4 DRD AB1  DH  -   -   -    ALW  -    DPTR -
[seq JMPIL]                     # JML [abs]
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 DRD AB0  DL  -   -   -    -    -    -    -
    4 DRD AB1  DH  -   -   -    -    -    -    -
    5 DRD AB2  BA  -   -   -    ALW  -    DPTRL -
[seq JMPIX]                     # JMP (abs,X)
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DRD PX0  DL  -   -   -    -    -    -    -
    5 DRD PX1  DH  -   -   -    ALW  -    DPTR -
[seq JSRIX]                     # JSR (abs,X): the pushed value is the address of the last byte
    1 PRG PC   AL  INC -   -    -    -    -    WIDE
    2 DWR SW   -   -   DEC PCH  -    -    -    WIDE
    3 DWR SW   -   -   DEC PCL  -    -    -    WIDE
    4 PRG PC   AH  INC -   -    -    -    -    WIDE
    5 IO  PCD  -   -   -   -    -    -    -    WIDE
    6 DRD PX0  DL  -   -   -    -    -    -    WIDE
    7 DRD PX1  DH  -   -   -    ALW  -    DPTR WIDE
[seq JSL]
    1 PRG PC   AL  INC -   -    -    -    -    WIDE
    2 PRG PC   AH  INC -   -    -    -    -    WIDE
    3 DWR SW   -   -   DEC PBR  -    -    -    WIDE
    4 IO  S1W  -   -   -   -    -    -    -    WIDE
    5 PRG PC   BA  INC -   -    -    -    -    WIDE
    6 DWR SW   -   -   DEC PCDH -    -    -    WIDE
    7 DWR SW   -   -   DEC PCDL ALW  -    LONG WIDE
[seq RTL]
    1 IO  PC   -   -   -   -    -    -    -    WIDE
    2 IO  PC   -   -   -   -    -    -    -    WIDE
    3 DRD S1W  AL  -   INC -    -    -    -    WIDE
    4 DRD S1W  AH  -   INC -    -    -    -    WIDE
    5 DRD S1W  BA  -   INC -    ALW  -    RTL  WIDE
[seq MVN]                       # MVN and MVP: one byte for each pass
    1 PRG PC   BA  INC -   -    -    -    -    -
    2 PRG PC   OF  INC -   -    -    -    -    -
    3 DRD MVS  DL  -   -   -    -    -    -    -
    4 DWR MVD  -   -   -   DL   -    -    -    -
    5 IO  MVD  -   -   -   -    -    -    -    -
    6 IO  MVD  -   -   -   -    ALW  -    REWIND -

# interrupts: BRK and COP read a signature byte; NMI and IRQ do not move the program counter.
# The program bank is pushed only in native mode.
[seq BRK]
    1 PRG PC   -   INC -   -    -    EMU  -    -
    2 DWR S    -   -   DEC PBR  -    -    -    -
    3 DWR S    -   -   DEC PCH  -    -    -    -
    4 DWR S    -   -   DEC PCL  -    -    -    -
    5 DWR S    -   -   DEC PSW  -    -    -    -
    6 DRD VEC  AL  -   -   -    -    -    -    -
    7 DRD VEC1 AH  -   -   -    ALW  -    VECT -
[seq INT]
    1 IO  PC   -   -   -   -    -    EMU  -    -
    2 DWR S    -   -   DEC PBR  -    -    -    -
    3 DWR S    -   -   DEC PCH  -    -    -    -
    4 DWR S    -   -   DEC PCL  -    -    -    -
    5 DWR S    -   -   DEC PSW  -    -    -    -
    6 DRD VEC  AL  -   -   -    -    -    -    -
    7 DRD VEC1 AH  -   -   -    ALW  -    VECT -
[seq RTI]                       # emulation mode ends at step 5
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 DRD S1   DL  -   INC -    -    -    -    -
    4 DRD S1   AL  -   INC -    -    -    -    -
    5 DRD S1   AH  -   INC -    EMU  -    ABS  -
    6 DRD S1   BA  -   INC -    ALW  -    LONG -
[seq WAI]                       # the third step repeats until an interrupt request
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 IO  PC   -   -   -   -    ALW  -    -    WAI
[seq STP]                       # the third step repeats until reset
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 IO  PC   -   -   -   -    ALW  -    -    STP
