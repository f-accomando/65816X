# c816_tables.txt - the only source of c816_decode and c816_ucode.
# Format and codes: docs/design/cpu-core.md, sections 4 and 5. After a change run: make tables

#   t bus asel cap pc  sop wsel last skip pcop hold
[seq IMP]                       # implied: 2 cycles
    1 IO  PC   -   -   -   -    ALW  -    -    -
[seq WDM]                       # WDM, ext_en = 0: 2 cycles, the second byte is skipped
    1 IO  PC   -   INC -   -    ALW  -    -    -
[seq WDMX]                      # WDM, ext_en = 1: the sub-opcode is a program byte
    1 PRG PC   DL  INC -   -    ALW  -    -    -
[seq IMM]                       # immediate: 2 (8-bit) or 3 (16-bit)
    1 PRG PC   DL  INC -   -    W8   -    -    -
    2 PRG PC   DH  INC -   -    ALW  -    -    -
[seq IMM8P]                     # REP, SEP: 3
    1 PRG PC   DL  INC -   -    -    -    -    -
    2 IO  PCD  -   -   -   -    ALW  -    -    -
[seq ABS]                       # 4 or 5
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 DAT EA   DL  -   -   RLO  W8   -    -    -
    4 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq ABSX]                      # load 4-6, store 5-6
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    XNC  -    -
    3 IO  DUMI -   -   -   -    -    -    -    -
    4 DAT EA   DL  -   -   RLO  W8   -    -    -
    5 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq LONG]                      # 5 or 6
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 PRG PC   BA  INC -   -    -    -    -    -
    4 DAT EA   DL  -   -   RLO  W8   -    -    -
    5 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq JMP]                       # 3
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    ALW  -    ABS  -
[seq JML]                       # 4
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 PRG PC   BA  INC -   -    ALW  -    LONG -
[seq BR]                        # 2 not taken, 3 taken, 4 taken with E = 1 and page cross
    1 PRG PC   AL  INC -   -    NTK  -    -    -
    2 IO  PCD  -   -   -   -    NEC  -    BRL  -
    3 IO  PCD  -   -   -   -    ALW  -    BR   -
[seq JSR]                       # 6
    1 PRG PC   AL  INC -   -    -    -    -    -
    2 PRG PC   AH  INC -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DWR S    -   -   DEC PCDH -    -    -    -
    5 DWR S    -   -   DEC PCDL ALW  -    ABS  -
[seq RTS]                       # 6
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 DRD S1   AL  -   INC -    -    -    -    -
    4 DRD S1   AH  -   INC -    -    -    -    -
    5 IO  S    -   -   -   -    ALW  -    RTS  -
[seq PUSH]                      # PHA: 3 or 4, high byte first
    1 IO  PC   -   -   -   -    -    W8   -    -
    2 DWR S    -   -   DEC RHI  -    -    -    -
    3 DWR S    -   -   DEC RLO  ALW  -    -    -
[seq PULL]                      # PLA: 4 or 5, low byte first
    1 IO  PC   -   -   -   -    -    -    -    -
    2 IO  PC   -   -   -   -    -    -    -    -
    3 DRD S1   DL  -   INC -    W8   -    -    -
    4 DRD S1   DH  -   INC -    ALW  -    -    -
[seq RST]                       # pseudo-op 100: simplified reset (C5 makes it exact)
    1 DRD VEC  AL  -   -   -    -    -    -    -
    2 DRD VEC1 AH  -   -   -    ALW  -    ABS  -

[seq DP]                        # direct page: 3 to 5, one more cycle if D[7:0] != 0
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DAT EA   DL  -   -   RLO  W8   -    -    -
    4 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPX]                       # direct page,X and ,Y
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DAT EA   DL  -   -   RLO  W8   -    -    -
    5 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPIND]                     # (dp)
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD DP0  AL  -   -   -    -    -    -    -
    4 DRD DP1  AH  -   -   -    -    -    -    -
    5 DAT EA   DL  -   -   RLO  W8   -    -    -
    6 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPXIND]                    # (dp,X)
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 IO  PCD  -   -   -   -    -    -    -    -
    4 DRD DX0  AL  -   -   -    -    -    -    -
    5 DRD DX1  AH  -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPINDY]                    # (dp),Y: the dummy cycle as in abs,X
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD DP0  AL  -   -   -    -    -    -    -
    4 DRD DP1  AH  -   -   -    -    XNC  -    -
    5 IO  DUMI -   -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq DPLONG]                    # [dp] and [dp],Y
    1 PRG PC   OF  INC -   -    -    DL0  -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD LP0  AL  -   -   -    -    -    -    -
    4 DRD LP1  AH  -   -   -    -    -    -    -
    5 DRD LP2  BA  -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq SR]                        # stack relative
    1 PRG PC   OF  INC -   -    -    -    -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DAT EA   DL  -   -   RLO  W8   -    -    -
    4 DAT EA1  DH  -   -   RHI  ALW  -    -    -
[seq SRINDY]                    # (sr,S),Y
    1 PRG PC   OF  INC -   -    -    -    -    -
    2 IO  PCD  -   -   -   -    -    -    -    -
    3 DRD SR0  AL  -   -   -    -    -    -    -
    4 DRD SR1  AH  -   -   -    -    -    -    -
    5 IO  SR1  -   -   -   -    -    -    -    -
    6 DAT EA   DL  -   -   RLO  W8   -    -    -
    7 DAT EA1  DH  -   -   RHI  ALW  -    -    -
